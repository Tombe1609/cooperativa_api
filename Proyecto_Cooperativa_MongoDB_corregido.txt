# Enunciado:
# Como estudiante quiero realizar un pago y seleccionar si deseo abonarlo
# al capital o al interés para controlar la forma en que disminuye mi deuda.

# IMPORTANTE:
# Este ejemplo utiliza transacciones, por lo que MongoDB debe ejecutarse
# como replica set. El pago y la actualización del préstamo se confirman juntos.

# ========================================================
# EJEMPLO TRANSACCIONAL DE ABONO AL CAPITAL
# ========================================================

# Préstamo P003:
# Capital pendiente inicial = $3.000.000
# Abono a capital = $1.000.000
# Nuevo capital = $2.000.000
# Nuevo interés pendiente = $2.000.000 * 0.05 = $100.000

const sesionCapitalHU8 = db.getMongo().startSession()
const dbCapitalHU8 = sesionCapitalHU8.getDatabase("cooperativa_universitaria")

try {
  sesionCapitalHU8.startTransaction()

  const prestamoCapitalHU8 = dbCapitalHU8.prestamos.findOne({
    prestamo_id: "P003",
    estado: "activo",
    capital_pendiente: { $gte: 1000000 }
  })

  if (!prestamoCapitalHU8) {
    throw new Error("Préstamo inexistente/inactivo o abono mayor al capital pendiente.")
  }

  const capitalNuevoHU8 = prestamoCapitalHU8.capital_pendiente - 1000000
  const interesNuevoHU8 = capitalNuevoHU8 * prestamoCapitalHU8.tasa_interes

  const actualizacionCapitalHU8 = dbCapitalHU8.prestamos.updateOne(
    {
      prestamo_id: "P003",
      estado: "activo",
      capital_pendiente: { $gte: 1000000 }
    },
    [
      {
        $set: {
          capital_pendiente: { $subtract: ["$capital_pendiente", 1000000] }
        }
      },
      {
        $set: {
          interes_pendiente: {
            $multiply: ["$capital_pendiente", "$tasa_interes"]
          }
        }
      },
      {
        $set: {
          estado: {
            $cond: [
              {
                $and: [
                  { $eq: ["$capital_pendiente", 0] },
                  { $eq: ["$interes_pendiente", 0] }
                ]
              },
              "pagado",
              "activo"
            ]
          }
        }
      }
    ]
  )

  if (actualizacionCapitalHU8.modifiedCount !== 1) {
    throw new Error("No se pudo actualizar el préstamo.")
  }

  dbCapitalHU8.pagos_prestamo.insertOne({
    pago_id: "PG004",
    prestamo_id: "P003",
    usuario_id: prestamoCapitalHU8.usuario_id,
    tipo_abono: "capital",
    valor: 1000000,
    fecha: ISODate("2026-03-18T10:00:00Z")
  })

  sesionCapitalHU8.commitTransaction()
  print("Abono a capital registrado correctamente.")
} catch (error) {
  sesionCapitalHU8.abortTransaction()
  print("Abono a capital cancelado: " + error.message)
} finally {
  sesionCapitalHU8.endSession()
}

db.prestamos.findOne(

# HISTORIA DE USUARIO 9
# ========================================================

# Enunciado:
# Como sistema quiero distribuir los intereses generados por los préstamos
# proporcionalmente entre los estudiantes ahorradores para reconocer su
# participación en el fondo de la cooperativa.

# Ejemplo:
#
# Total ahorrado = $27.000.000
# Intereses generados = $500.000
#
# participación = ahorro_estudiante / total_ahorrado
# interes_recibido = participación * intereses_generados

# --------------------------------------------------------
# AGREGACIÓN DE DISTRIBUCIÓN
# --------------------------------------------------------

db.ahorros.aggregate([
  {
    $match: {
      estado: "activo",
      saldo: { $gt: 0 }
    }
  },
  {
    $group: {
      _id: null,
      total_ahorrado: { $sum: "$saldo" },
      cuentas: {
        $push: {
          ahorro_id: "$ahorro_id",
          usuario_id: "$usuario_id",
          saldo: "$saldo"
        }
      }
    }
  },
  {
    $unwind: "$cuentas"
  },
  {
    $project: {
      _id: 0,
      ahorro_id: "$cuentas.ahorro_id",
      usuario_id: "$cuentas.usuario_id",
      ahorro_estudiante: "$cuentas.saldo",
      total_ahorrado: 1,
      intereses_generados: { $literal: 500000 },
      participacion: {
        $divide: [
          "$cuentas.saldo",
          "$total_ahorrado"
        ]
      }
    }
  },
  {
    $set: {
      interes_recibido: {
        $multiply: [
          "$participacion",
          "$intereses_generados"
        ]
      }
    }
  },
  {
    $project: {
      ahorro_id: 1,
      usuario_id: 1,
      ahorro_estudiante: 1,
      total_ahorrado: 1,
      intereses_generados: 1,
      participacion: 1,
      interes_recibido: {
        $round: ["$interes_recibido", 2]
      }
    }
  }
])
